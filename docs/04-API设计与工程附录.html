错误分级与处理策略

级别场景策略HTTP 状态码
致命模型加载失败 / Qdrant 连接断开返回 503，记录告警503

降级策略

class DegradationHandler:
    """模态降级处理器 — 部分模态失败时保证基本可用"""

    DEGRADATION_CHAIN = {
        "video": {
            "vlm_failed":      "skip_visual",    # VLM 挂了 → 跳过画面，仅索引声音
            "asr_failed":      "skip_audio",     # ASR 挂了 → 跳过语音，仅索引画面
            "clap_failed":     "skip_clap",      # CLAP 挂了 → 跳过音频嵌入
            "all_visual_fail": "text_only",      # 画面全挂 → 仅用文件名 + 元数据
        },
        "audio": {
            "asr_failed":      "clap_only",      # ASR 挂了 → 仅用 CLAP 嵌入
            "clap_failed":     "asr_only",       # CLAP 挂了 → 仅用 ASR 文本
            "all_fail":        "metadata_only",  # 全挂 → 仅用文件名
        },
        "search": {
            "reranker_failed": "skip_rerank",    # Reranker 挂了 → 直接用向量检索结果
            "llm_failed":      "skip_explain",   # LLM 挂了 → 返回无推荐理由的结果
        }
    }

    async def handle_pipeline_error(
        self, media_type: str, failed_component: str, context: dict
    ) -> dict:
        """根据失败组件选择降级路径"""
        chain = self.DEGRADATION_CHAIN.get(media_type, {})
        action = chain.get(failed_component)

        if action is None:
            raise PipelineError(f"无法降级：{media_type}.{failed_component}")

        logger.warning(
            f"降级触发: {media_type}/{failed_component} → {action}, "
            f"file={context.get('file_path')}"
        )

        return {"degraded": True, "action": action, "original_error": failed_component}

重试机制

import asyncio
import random
from functools import wraps

def retry_with_backoff(
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
    exceptions: tuple = (TimeoutError, ConnectionError, OOMError),
):
    """指数退避 + 抖动重试装饰器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            for attempt in range(max_retries + 1):
                try:
                    return await func(*args, **kwargs)
                except exceptions as e:
                    if attempt == max_retries:
                        raise
                    delay = min(
                        base_delay * (2 ** attempt) + random.uniform(0, 1),
                        max_delay
                    )
                    logger.warning(
                        f"重试 {attempt+1}/{max_retries}: {func.__name__}, "
                        f"error={e}, delay={delay:.1f}s"
                    )
                    await asyncio.sleep(delay)
        return wrapper
    return decorator

# 使用示例
@retry_with_backoff(max_retries=3, base_delay=2.0)
async def vlm_analyze(image_path: str) -> str:
    """VLM 分析（自动重试）"""
    return await vlm_model.chat(image_path, prompt=ANALYZE_PROMPT)

04批量处理与并发控制

并发限制原则

24GB 内存环境下，不同模态的处理内存占用差异巨大。通过信号量严格控制并发数，防止 OOM。

处理类型并发上限内存峰值/任务总内存预算信号量
图片 VLM 分析5~1.5 GB7.5 GBSemaphore(5)
音频 ASR 转写3~0.8 GB2.4 GBSemaphore(3)
音频 CLAP 嵌入3~0.6 GB1.8 GBSemaphore(3)
视频处理（抽帧+VLM）1~3.0 GB3.0 GBSemaphore(1)
Embedding 向量化10~0.3 GB3.0 GBSemaphore(10)
 
import asyncio
from dataclasses import dataclass
from typing import Callable, Any

@dataclass
class BatchProcessor:
    """批量处理器 — 信号量限流 + 进度追踪"""

    # 模态并发限制
    SEMAPHORES = {
        "image":    asyncio.Semaphore(5),
        "audio":    asyncio.Semaphore(3),
        "video":    asyncio.Semaphore(1),
        "embedding": asyncio.Semaphore(10),
    }

    async def process_batch(
        self,
        files: list[dict],           # [{"path": "...", "type": "image", ...}, ...]
        processor: Callable,          # 处理函数
        progress_callback: Callable = None,
    ) -> list[dict]:
        """批量处理文件，按模态并发"""
        total = len(files)
        completed = 0
        failed = 0
        results = []

        async def process_one(file_info: dict) -> dict:
            nonlocal completed, failed
            sem = self.SEMAPHORES.get(file_info["type"], asyncio.Semaphore(1))
            try:
                async with sem:
                    result = await processor(file_info)
                    completed += 1
                    if progress_callback:
                        await progress_callback(completed, total, failed)
                    return {"status": "ok", "file": file_info["path"], "result": result}
            except Exception as e:
                failed += 1
                logger.error(f"处理失败: {file_info['path']}, error={e}")
                return {"status": "error", "file": file_info["path"], "error": str(e)}

        tasks = [process_one(f) for f in files]
        results = await asyncio.gather(*tasks, return_exceptions=False)
        return results

class ProgressiveIndexer:
    """渐进式索引器 — L1→L2→L3 逐层加深"""

    PRIORITY_QUEUE = asyncio.PriorityQueue()

    async def index_l1(self, file_path: str) -> str:
        """L1 快速索引 — 立即可搜索"""
        item_id = str(uuid4())
        metadata = extract_file_metadata(file_path)  # 文件名/大小/分辨率

        await qdrant.upsert(
            collection="media_items",
            point_id=item_id,
            vector={  # 占位向量，L2 替换
                "content_vector": [0.0] * 2048,
            },
            payload={
                "id": item_id,
                "file_path": file_path,
                "file_hash": file_md5(file_path),
                "media_type": metadata["media_type"],
                "file_size": metadata["file_size"],
                "resolution": metadata.get("resolution"),
                "index_level": 1,          # 标记索引深度
                "created_at": datetime.now().isoformat(),
            }
        )

        # 排队 L2 标准索引
        await self.PRIORITY_QUEUE.put((2, item_id, file_path))
        return item_id

    async def index_l2(self, item_id: str, file_path: str):
        """L2 标准索引 — VLM + Embedding"""
        description = await vlm_analyze(file_path)
        tags = extract_tags(description)
        vector = await embedding_model.encode(description)

        await qdrant.set_payload(
            collection="media_items",
            point_id=item_id,
            payload={
                "description": description,
                "tags_category": tags["category"],
                "tags_style": tags["style"],
                "tags_emotion": tags["emotion"],
                "index_level": 2,
            }
        )
        await qdrant.update_vectors(
            collection="media_items",
            point_id=item_id,
            vectors={"content_vector": vector},
        )

        # 视频/音频排队 L3 深度索引
        media_type = get_media_type(file_path)
        if media_type in ("video", "audio"):
            await self.PRIORITY_QUEUE.put((3, item_id, file_path))

    async def index_l3(self, item_id: str, file_path: str):
        """L3 深度索引 — ASR + CLAP（仅音视频）"""
        tasks = []

        # ASR 转写
        if has_audio_track(file_path):
            tasks.append(self._do_asr(item_id, file_path))

        # CLAP 音频嵌入
        if has_audio_track(file_path):
            tasks.append(self._do_clap(item_id, file_path))

        await asyncio.gather(*tasks)

        await qdrant.set_payload(
            collection="media_items",
            point_id=item_id,
            payload={"index_level": 3}
        )

    async def run_worker(self):
        """后台 worker — 按优先级处理 L2/L3 队列"""
        while True:
            priority, item_id, file_path = await self.PRIORITY_QUEUE.get()
            try:
                if priority == 2:
                    await self.index_l2(item_id, file_path)
                elif priority == 3:
                    await self.index_l3(item_id, file_path)
            except Exception as e:
                logger.error(f"渐进索引失败 L{priority}: {item_id}, {e}")
                # 失败重排队（降低优先级）
                await self.PRIORITY_QUEUE.put((priority + 0.5, item_id, file_path))

06评估指标与反馈闭环

离线评估指标

检索质量

  • Recall@K：Top-K 中命中相关条目的比例
  • Precision@K：Top-K 中相关条目的占比
  • MRR：第一个相关结果的平均倒数排名
  • NDCG@K：考虑排名位置的增益（推荐主指标）

推荐质量

  • Coverage：被推荐条目占总条目的比例
  • Novelty：推荐长尾内容的能力
  • Diversity：推荐列表内的多样性（ILS）
  • Serpent：推荐列表的惊喜度

评估脚本

# scripts/evaluate.py

import numpy as np
from typing import list

def recall_at_k(retrieved: list[str], relevant: set[str], k: int) -> float:
    """Recall@K — Top-K 中命中的相关条目比例"""
    top_k = retrieved[:k]
    hits = len(set(top_k) & relevant)
    return hits / len(relevant) if relevant else 0.0

def ndcg_at_k(retrieved: list[str], relevant: set[str], k: int) -> float:
    """NDCG@K — 归一化折损累积增益"""
    dcg = 0.0
    for i, item in enumerate(retrieved[:k]):
        rel = 1.0 if item in relevant else 0.0
        dcg += rel / np.log2(i + 2)

    ideal_len = min(len(relevant), k)
    idcg = sum(1.0 / np.log2(i + 2) for i in range(ideal_len))
    return dcg / idcg if idcg > 0 else 0.0

def intra_list_similarity(items: list[dict], vectors: dict[str, list[float]]) -> float:
    """ILS — 推荐列表内平均余弦相似度（越低越多样）"""
    vecs = [vectors[item["item_id"]] for item in items if item["item_id"] in vectors]
    if len(vecs) < 2:
        return 0.0
    sims = []
    for i in range(len(vecs)):
        for j in range(i + 1, len(vecs)):
            sim = cosine_similarity(vecs[i], vecs[j])
            sims.append(sim)
    return np.mean(sims)

class ColdStartHandler:
    """冷启动处理器"""

    # 新条目 boost 配置
    NEW_ITEM_BOOST = {
        "weight": 1.2,        # 分数加权
        "decay_days": 7,      # 7 天线性衰减至 1.0
    }

    def apply_new_item_boost(
        self, results: list[dict], items_meta: dict[str, dict]
    ) -> list[dict]:
        """对新条目（入库 <7 天）施加 boost"""
        now = datetime.now()
        for item in results:
            meta = items_meta.get(item["item_id"])
            if meta and "created_at" in meta:
                age_days = (now - parse(meta["created_at"])).days
                if age_days < self.NEW_ITEM_BOOST["decay_days"]:
                    decay = 1 - (age_days / self.NEW_ITEM_BOOST["decay_days"])
                    boost = 1 + (self.NEW_ITEM_BOOST["weight"] - 1) * decay
                    item["score"] *= boost
                    item["boosted"] = True

        results.sort(key=lambda x: -x["score"])
        return results

08MMR 多样性控制

Maximal Marginal Relevance (MMR) 算法在精排后对推荐列表去重和多样化，避免连续推荐高度相似的内容。

MMR 核心公式

MMR(d) = argmaxd ∈ C\D [ λ · sim(d, q) − (1−λ) · maxd' ∈ D sim(d, d') ]

  • λ = 1.0：纯相关性排序（无多样性）
  • λ = 0.5：相关性与多样性均衡（默认推荐）
  • λ = 0.2：强调多样性（探索性推荐）
import numpy as np
from typing import Sequence

def maximal_marginal_relevance(
    query_vector: np.ndarray,
    candidate_vectors: np.ndarray,
    candidate_ids: Sequence[str],
    lambda_param: float = 0.5,
    k: int = 10,
) -> list[str]:
    """
    MMR 多样性重排

    Args:
        query_vector: 查询向量 (D,)
        candidate_vectors: 候选向量矩阵 (N, D)
        candidate_ids: 候选 ID 列表 (N,)
        lambda_param: 相关性 vs 多样性权衡
        k: 返回数量

    Returns:
        重排后的 item_id 列表
    """
    # 归一化
    query_norm = query_vector / (np.linalg.norm(query_vector) + 1e-8)
    cand_norm = candidate_vectors / (
        np.linalg.norm(candidate_vectors, axis=1, keepdims=True) + 1e-8
    )

    # 查询-候选相似度
    rel_scores = cand_norm @ query_norm  # (N,)

    selected = []
    selected_mask = np.zeros(len(candidate_ids), dtype=bool)

    for _ in range(min(k, len(candidate_ids))):
        best_idx = -1
        best_score = -np.inf

        for i in range(len(candidate_ids)):
            if selected_mask[i]:
                continue

            # 相关性分数
            relevance = rel_scores[i]

            # 多样性惩罚：与已选中最相似的分数
            if selected:
                diversity = max(
                    float(cand_norm[i] @ cand_norm[j]) for j in selected
                )
            else:
                diversity = 0.0

            mmr_score = lambda_param * relevance - (1 - lambda_param) * diversity

            if mmr_score > best_score:
                best_score = mmr_score
                best_idx = i

        if best_idx == -1:
            break

        selected.append(best_idx)
        selected_mask[best_idx] = True

    return [candidate_ids[i] for i in selected]


# 在推荐引擎中的集成
class RecommendationEngine:
    async def recommend(self, query: dict, top_n: int = 10) -> list[dict]:
        # 1. 向量检索 Top-100
        candidates = await self.searcher.search(query, top_k=100)

        # 2. Reranker 精排 → Top-20
        reranked = await self.reranker.rerank(query, candidates, top_k=20)

        # 3. MMR 多样性重排 → Top-N
        candidate_vectors = np.array([c["vector"] for c in reranked])
        candidate_ids = [c["item_id"] for c in reranked]
        query_vec = np.array(query["vector"])

        mmr_ids = maximal_marginal_relevance(
            query_vector=query_vec,
            candidate_vectors=candidate_vectors,
            candidate_ids=candidate_ids,
            lambda_param=query.get("mmr_lambda", 0.5),
            k=top_n,
        )

        # 4. LLM 生成推荐理由
        final = [c for c in reranked if c["item_id"] in mmr_ids]
        final = await self.explainer.explain(query, final)

        return final
